        academyTerm: {
          on: { TERM_COMPLETE: 'graduation' }

        graduatedHonours: {
          on: { CONTINUE: '#creation.career' }
        },
        failedGraduation: {
          on: {
            RETRY: { target: 'choosing', guard: 'canRetryEducation' },
            CONTINUE: '#creation.career'
          }
        }
      }
    }
